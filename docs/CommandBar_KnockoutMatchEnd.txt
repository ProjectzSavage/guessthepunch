local function ensureFolder(parent, name)
	local f = parent:FindFirstChild(name)
	if not f then
		f = Instance.new("Folder")
		f.Name = name
		f.Parent = parent
		print("Created Folder: " .. parent:GetFullName() .. "." .. name)
	end
	return f
end

local function ensureModule(parent, name)
	local m = parent:FindFirstChild(name)
	if not m then
		m = Instance.new("ModuleScript")
		m.Name = name
		m.Parent = parent
		print("Created ModuleScript: " .. parent:GetFullName() .. "." .. name)
	end
	return m
end

local function ensureRemote(parent, name)
	local r = parent:FindFirstChild(name)
	if not r then
		r = Instance.new("RemoteEvent")
		r.Name = name
		r.Parent = parent
		print("Created RemoteEvent: " .. parent:GetFullName() .. "." .. name)
	end
	return r
end

-- OPTIONAL per-arena SpectatorArea (server auto-creates a floating platform if missing)
local Workspace = game:GetService("Workspace")
local MatchArenas = Workspace:FindFirstChild("MatchArenas")
if MatchArenas then
	for _, arena in ipairs(MatchArenas:GetChildren()) do
		if arena:IsA("Model") and not arena:FindFirstChild("SpectatorArea") then
			local ok, cf = pcall(function()
				local bcf, bsize = arena:GetBoundingBox()
				return CFrame.new(bcf.Position + Vector3.new(0, (bsize.Y / 2) + 25, 0))
			end)
			if ok and cf then
				local platform = Instance.new("Part")
				platform.Name = "SpectatorArea"
				platform.Anchored = true
				platform.CanCollide = true
				platform.Transparency = 0.4
				platform.Material = Enum.Material.Glass
				platform.Color = Color3.fromRGB(120, 160, 220)
				platform.Size = Vector3.new(24, 1, 24)
				platform.CFrame = cf
				platform.Parent = arena
				print("Created SpectatorArea platform in " .. arena.Name)
			end
		end
	end
else
	warn("Workspace.MatchArenas not found - server will auto-create platforms per match")
end
