-- Command Bar Script for Guess the Puncher - Shared Config Only
-- Purpose: Creates folders and ModuleScripts with correct names (no .luau extensions)
-- How to use: Open Roblox Studio > View > Command Bar > Paste this entire script > Press Enter
-- Then manually paste code from GitHub src/Shared/ into each ModuleScript

local ReplicatedStorage = game:GetService("ReplicatedStorage")

-- Create Shared folder
local sharedFolder = ReplicatedStorage:FindFirstChild("Shared")
if not sharedFolder then
    sharedFolder = Instance.new("Folder")
    sharedFolder.Name = "Shared"
    sharedFolder.Parent = ReplicatedStorage
    print("Created ReplicatedStorage.Shared")
else
    print("Found existing ReplicatedStorage.Shared")
end

-- List of ModuleScript names to create (no extensions)
local moduleNames = {
    "Config",
    "MatchStates",
    "Roles",
    "PowerZones",
    "Results",
    "Types",
    "Constants"
}

for _, name in ipairs(moduleNames) do
    local existing = sharedFolder:FindFirstChild(name)
    if not existing then
        local mod = Instance.new("ModuleScript")
        mod.Name = name
        mod.Parent = sharedFolder
        mod.Source = "-- " .. name .. ".luau\n-- Paste code from GitHub src/Shared/" .. name .. ".luau here\nreturn {}\n"
        print("Created ModuleScript: Shared/" .. name)
    else
        if existing:IsA("ModuleScript") then
            print("Already exists: Shared/" .. name .. " (ModuleScript)")
        else
            warn("Object named " .. name .. " exists but is not ModuleScript: " .. existing.ClassName)
        end
    end
end

print("Done! Now copy code from GitHub src/Shared/ into each ModuleScript.")
print("Verify with: require(ReplicatedStorage.Shared.Config) etc.")
