-- CommandBar script to create Puncher Selection system
-- Paste into Studio Command Bar and run

local function ensureFolder(parent, name)
    local folder = parent:FindFirstChild(name)
    if not folder then
        folder = Instance.new("Folder")
        folder.Name = name
        folder.Parent = parent
        print("Created Folder: " .. parent:GetFullName() .. "." .. name)
    end
    return folder
end

local function ensureModuleScript(parent, name)
    local mod = parent:FindFirstChild(name)
    if not mod then
        mod = Instance.new("ModuleScript")
        mod.Name = name
        mod.Parent = parent
        print("Created ModuleScript: " .. parent:GetFullName() .. "." .. name)
    end
    return mod
end

-- Remotes
local Remotes = ensureFolder(ReplicatedStorage, "Remotes")
local function ensureRemote(name)
    local re = Remotes:FindFirstChild(name)
    if not re then
        re = Instance.new("RemoteEvent")
        re.Name = name
        re.Parent = Remotes
        print("Created RemoteEvent: Remotes." .. name)
    end
    return re
end

-- Client
local StarterPlayer = game:GetService("StarterPlayer")
local StarterPlayerScripts = StarterPlayer:FindFirstChild("StarterPlayerScripts") or ensureFolder(StarterPlayer, "StarterPlayerScripts")
local GuessThePuncherClient = StarterPlayerScripts:FindFirstChild("GuessThePuncherClient")
if not GuessThePuncherClient then
    GuessThePuncherClient = Instance.new("LocalScript")
    GuessThePuncherClient.Name = "GuessThePuncherClient"
    GuessThePuncherClient.Parent = StarterPlayerScripts
    print("Created LocalScript: GuessThePuncherClient")
end
local ClientServices = GuessThePuncherClient:FindFirstChild("Services")
if not ClientServices then
    ClientServices = Instance.new("Folder")
    ClientServices.Name = "Services"
    ClientServices.Parent = GuessThePuncherClient
    print("Created Folder: GuessThePuncherClient.Services")
end
local TargetCameraService = ensureModuleScript(ClientServices, "TargetCameraService")
local TargetVisibilityService = ensureModuleScript(ClientServices, "TargetVisibilityService")
local PuncherSelectionService = ensureModuleScript(ClientServices, "PuncherSelectionService")

-- Arena check
local Workspace = game:GetService("Workspace")
local MatchArenas = Workspace:FindFirstChild("MatchArenas")
if MatchArenas then
    for _, arena in ipairs(MatchArenas:GetChildren()) do
        if arena:IsA("Model") then
            local targetStand = arena:FindFirstChild("TargetStand")
            local puncherStand = arena:FindFirstChild("PuncherStand")
            if not targetStand then
                warn("Missing TargetStand in " .. arena.Name .. " - create Part named TargetStand")
            end
            if not puncherStand then
                warn("Missing PuncherStand in " .. arena.Name .. " - create Part named PuncherStand behind TargetStand at -LookVector*6")
                -- Auto-create for testing
                if targetStand and targetStand:IsA("BasePart") then
                    local ps = Instance.new("Part")
                    ps.Name = "PuncherStand"
                    ps.Anchored = true
                    ps.CanCollide = false
                    ps.Transparency = 0.5
                    ps.Size = Vector3.new(4,1,4)
                    ps.CFrame = targetStand.CFrame * CFrame.new(0,0,6)
                    ps.Parent = arena
                    print("Auto-created PuncherStand in " .. arena.Name .. " at " .. tostring(ps.CFrame))
                end
            end
        end
    end
else
    warn("Missing Workspace.MatchArenas folder - create per docs")
end
