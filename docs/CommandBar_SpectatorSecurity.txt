local function ensureFolder(parent, name)
	local f = parent:FindFirstChild(name)
	if not f then
		f = Instance.new("Folder")
		f.Name = name
		f.Parent = parent
		print("Created Folder: " .. parent:GetFullName() .. "." .. name)
	end
	return f
end

local function ensureModule(parent, name)
	local m = parent:FindFirstChild(name)
	if not m then
		m = Instance.new("ModuleScript")
		m.Name = name
		m.Parent = parent
		print("Created ModuleScript: " .. parent:GetFullName() .. "." .. name)
	end
	return m
end

local function ensureRemote(parent, name)
	local r = parent:FindFirstChild(name)
	if not r then
		r = Instance.new("RemoteEvent")
		r.Name = name
		r.Parent = parent
		print("Created RemoteEvent: " .. parent:GetFullName() .. "." .. name)
	end
	return r
end
