for _, name in ipairs(sharedModules) do

        m.Name = name

end

remotesFolder.Name = "Remotes"
remotesFolder.Parent = ReplicatedStorage

matchArenasFolder.Name = "MatchArenas"
matchArenasFolder.Parent = Workspace

-- Helper to create Slot/Arena with requested hierarchy: TargetStand, PuncherStand, PlayerSpawns/Player1,Player2
local function createSlot(parentFolder, slotName, playerCount)
    local slotModel = parentFolder:FindFirstChild(slotName)
    if not slotModel then
        slotModel = Instance.new("Model")
        slotModel.Name = slotName
        slotModel.Parent = parentFolder
        print("Created " .. parentFolder.Name .. "/" .. slotName)
    end

    -- TargetStand
    if not slotModel:FindFirstChild("TargetStand") then
        local part = Instance.new("Part")
        part.Name = "TargetStand"
        part.Size = Vector3.new(4,1,4)
        part.Anchored = true
        part.Parent = slotModel
        print("  Created " .. slotName .. "/TargetStand")
    end

    -- PuncherStand (6 studs behind TargetStand)
    if not slotModel:FindFirstChild("PuncherStand") then
        local part = Instance.new("Part")
        part.Name = "PuncherStand"
        part.Size = Vector3.new(4,1,4)
        part.Anchored = true
        part.Parent = slotModel
        print("  Created " .. slotName .. "/PuncherStand")
    end

    -- PlayerSpawns Folder
    local spawnsFolder = slotModel:FindFirstChild("PlayerSpawns")
    if not spawnsFolder then
        spawnsFolder = Instance.new("Folder")
        spawnsFolder.Name = "PlayerSpawns"
        spawnsFolder.Parent = slotModel
        print("  Created " .. slotName .. "/PlayerSpawns")
    end

    for i=1, playerCount do
        local spawnName = "Player" .. i
        if not spawnsFolder:FindFirstChild(spawnName) then
            local part = Instance.new("Part")
            part.Name = spawnName
            part.Size = Vector3.new(3,1,3)
            part.Anchored = true
            part.Parent = spawnsFolder
            print("    Created " .. slotName .. "/PlayerSpawns/" .. spawnName)
        end
    end

    -- For lobby slots, also TouchBlock and DisplayBoard
    if parentFolder.Name == "MatchSlots" then
        if not slotModel:FindFirstChild("TouchBlock") then
            local part = Instance.new("Part")
            part.Name = "TouchBlock"
            part.Size = Vector3.new(6,1,6)
            part.Anchored = true
            part.CanCollide = false
            part.Parent = slotModel
            print("  Created " .. slotName .. "/TouchBlock")
        end
        if not slotModel:FindFirstChild("DisplayBoard") then
            local part = Instance.new("Part")
            part.Name = "DisplayBoard"
            part.Size = Vector3.new(4,6,0.5)
            part.Anchored = true
            part.Parent = slotModel
            print("  Created " .. slotName .. "/DisplayBoard")
        end
    end
end

-- Create Lobby Slots (queue)
createSlot(matchSlotsFolder, "Slot2", 2)
createSlot(matchSlotsFolder, "Slot4", 4)
createSlot(matchSlotsFolder, "Slot6", 6)

-- Create Match Arenas (actual match) with same hierarchy
createSlot(matchArenasFolder, "Arena2", 2)
createSlot(matchArenasFolder, "Arena4", 4)
createSlot(matchArenasFolder, "Arena6", 6)

    gui.Name = "MainHUD"
    gui.ResetOnSpawn = false
    gui.Parent = StarterGui
    print("Created StarterGui/MainHUD")
end
